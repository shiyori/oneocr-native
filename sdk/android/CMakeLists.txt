cmake_minimum_required(VERSION 3.22)
project(oneocr_android LANGUAGES CXX)
if(NOT ANDROID)
  message(FATAL_ERROR "Build with the Android NDK toolchain")
endif()
set(ONEOCR_GO_LIBRARY "" CACHE FILEPATH "Go c-shared liboneocr.so for ANDROID_ABI")
if(NOT EXISTS "${ONEOCR_GO_LIBRARY}")
  message(FATAL_ERROR "Set ONEOCR_GO_LIBRARY to the Go Android build output")
endif()
add_library(oneocr SHARED IMPORTED)
set_target_properties(oneocr PROPERTIES IMPORTED_LOCATION "${ONEOCR_GO_LIBRARY}")
add_library(oneocr_jni SHARED jni/oneocr_jni.cpp)
target_compile_features(oneocr_jni PRIVATE cxx_std_17)
target_compile_options(oneocr_jni PRIVATE -Wall -Wextra -Werror)
target_include_directories(oneocr_jni PRIVATE ../include)
target_link_libraries(oneocr_jni PRIVATE oneocr)
target_link_options(oneocr_jni PRIVATE "-Wl,-z,max-page-size=16384" "-Wl,-z,common-page-size=16384")
