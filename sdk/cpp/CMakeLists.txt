cmake_minimum_required(VERSION 3.22)
project(oneocr_cpp LANGUAGES CXX)
set(ONEOCR_LIBRARY "" CACHE FILEPATH "Path to liboneocr shared library (Windows: import library)")
if(NOT EXISTS "${ONEOCR_LIBRARY}")
  message(FATAL_ERROR "Set -DONEOCR_LIBRARY=/absolute/path/to/liboneocr.dylib or .so/.lib")
endif()
add_executable(oneocr-cpp main.cpp)
target_compile_features(oneocr-cpp PRIVATE cxx_std_17)
if(MINGW)
  target_link_options(oneocr-cpp PRIVATE -municode)
endif()
if(MSVC)
  target_compile_options(oneocr-cpp PRIVATE /utf-8)
endif()
target_include_directories(oneocr-cpp PRIVATE ../include)
target_link_libraries(oneocr-cpp PRIVATE "${ONEOCR_LIBRARY}")
get_filename_component(ONEOCR_LIBRARY_DIR "${ONEOCR_LIBRARY}" DIRECTORY)
set_target_properties(oneocr-cpp PROPERTIES BUILD_RPATH "${ONEOCR_LIBRARY_DIR}")
include(CTest)
if(BUILD_TESTING)
  add_executable(oneocr-abi-test abi_test.cpp)
  target_compile_features(oneocr-abi-test PRIVATE cxx_std_17)
  target_include_directories(oneocr-abi-test PRIVATE ../include)
  target_link_libraries(oneocr-abi-test PRIVATE "${ONEOCR_LIBRARY}")
  set_target_properties(oneocr-abi-test PROPERTIES BUILD_RPATH "${ONEOCR_LIBRARY_DIR}")
  add_test(NAME abi-errors COMMAND oneocr-abi-test)
endif()
